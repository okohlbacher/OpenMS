            )
            for name in FILES
        },

def bind_identity():
    payload = identity()
    path = OUT / "measurement-identity.json"
    if path.exists():
        assert (
            json.loads(path.read_text()) == payload
        ), "Measurement identity changed: refuse mixed-config resume"

        path.write_text(json.dumps(payload, indent=2) + "\n")
